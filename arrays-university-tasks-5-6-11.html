05

«Плотный» массив

Является ли отсортированный массив «плотным» (соседи отличаются не более чем на 1)?

Условие: Прочитать массив целых чисел из input.txt. Отсортировать по возрастанию. Проверить, является ли он «плотным» (соседние элементы отличаются не более чем на 1). Вывести 1 (да) или 0 (нет).

Определение и логика

Плотный массив — неубывающий массив без «дыр»: для всех соседних пар a[i] и a[i+1] выполняется a[i+1] - a[i] ≤ 1.

Примеры плотных: [1, 1, 2, 2, 3] (разности 0, 1, 0, 1); [5, 5, 5] (все разности 0).

Примеры не плотных: [1, 3, 5] (разности 2, 2); [1, 2, 4] (разность 2 между индексами 1 и 2).

Пример трассировки

Входные данные: 3, 1, 3, 2. Ожидаемый результат: 1.

ШагОписаниеЗначение переменных
ЧтениеМассив прочитан из файлаa[0]=1, a[1]=3, a[2]=2
После сортировкиВызов sort() упорядочивает элементыa[0]=1, a[1]=2, a[2]=3

Полная программа

#include<stdio.h>
#include<math.h>
#include<stdlib.h>
#include<stdint.h>
static void sort(int *a, int n){
    for(int i = 0; i < n-1; i++){
        for(int j =  i + 1; j < n; j++){
            if(a[i] > a[j] ){
                int m = a[i];
                a[i] = a[j];
                a[j] = m;
            } 
        }
    }
}
int main(void)
{
    FILE *fin = fopen("input.txt", "r");
    if (fin == NULL) return -1;
    FILE *fout = fopen("output.txt", "w");
    if (fout == NULL) { fclose(fin); return -1; }

    int status = -1;
    int n = 0;
    int *a = NULL;
    if (fscanf(fin, "%d", &n) != 1 || n <= 0) goto done;
    if ((size_t)n > SIZE_MAX / sizeof *a) goto done;
    a = malloc((size_t)n * sizeof *a);
    if (a == NULL) goto done;
    for (int i = 0; i < n; ++i)
        if (fscanf(fin, "%d", &a[i]) != 1) goto done;

    sort(a, n);
    int dense = 1;
   
    for (int i = 1; i < n; ++i) {
        /* Приводим к long long, чтобы избежать переполнения int. */
        long long gap = (long long)a[i] - a[i - 1];
        if (gap > 1) dense = 0;
    }
    if (fprintf(fout, "%d\n", dense) < 0) goto done;
    status = 0;

done:
    free(a);
    if (fclose(fin) != 0) status = -1;
    if (fclose(fout) != 0) status = -1;
    return status;
}

Как объяснить этот код

  1. Функция sort: сортирует массив так же, как в задаче 1. Используется алгоритм с двумя вложенными циклами (пузырёк).
  2. Переменная dense: флаг, изначально равный 1 (да, плотный). Если на каком-то шаге находится разность больше 1, флаг становится 0.
  3. Цикл проверки: проходит по всем соседним парам, начиная с индекса 1. Вычисляет gap = a[i] - a[i-1].
  4. Переполнение: выражение приводится к long long перед вычитанием, чтобы избежать переполнения типа int при работе с крайними значениями.
  5. Вывод: если цикл завершился и dense == 1, массив плотный; иначе в файл пишется 0.
КЛЮЧЕВОЙ МОМЕНТСначала сортируем, потом проверяем. Только отсортированный массив имеет смысл для проверки плотности.
06

«Счастливый» массив

Существует ли точка раздела, где сумма левой части равна сумме правой?

Условие: Прочитать массив целых чисел. Существует ли его разбиение на две непустые части такие, что сумма левой части равна сумме правой? Вывести 1 (да) или 0 (нет).

Идея алгоритма

Массив «счастлив», если существует индекс k (1 ≤ k < n) такой, что:

a[0] + a[1] + ... + a[k-1] = a[k] + a[k+1] + ... + a[n-1]

Эквивалентно: sum_left = sum_right или sum_left = total_sum - sum_left, откуда 2 * sum_left = total_sum.

Пример

Входные данные: 4, 2, 3, 1, 4. Ожидаемый результат: 1.

ШагkЛевая частьПравая частьРезультат
Всего—[2, 3][1, 4]Обе суммы = 5 ✓
Проверкаk=22 + 3 = 51 + 4 = 5Массив счастлив!

Полная программа

#include<stdio.h>
#include<stdlib.h>
#include<stdint.h>
int happy(int *a, int n){
    int total = 0;
    for(int i = 0; i < n; i++){
        total += a[i];
    }
    int left = 0;
    for(int j = 1; j < n; ++j){
        left += a[j-1];
        if (left == total - left)
            return 1;
    }
    return 0;
}
int main(void)
{
    FILE *fin = fopen("input.txt", "r");
    if (fin == NULL)
        return -1;

    FILE *fout = fopen("output.txt", "w");
    if (fout == NULL) {
        fclose(fin);
        return -1;
    }

    int status = -1;
    int n = 0;
    int *a = NULL;

    if (fscanf(fin, "%d", &n) != 1 || n <= 0)
        goto done;

    if ((size_t)n > SIZE_MAX / sizeof *a)
        goto done;

    a = malloc((size_t)n * sizeof *a);
    if (a == NULL)
        goto done;

    for (int i = 0; i < n; ++i) {
        if (fscanf(fin, "%d", &a[i]) != 1)
            goto done;
    }

    if (fprintf(fout, "%d\n", happy(a, n)) < 0)
        goto done;

    status = 0;

done:
    free(a);
    if (fclose(fin) != 0)
        status = -1;
    if (fclose(fout) != 0)
        status = -1;

    return status;
}

Как объяснить этот код

  1. Первый цикл: вычисляет общую сумму total всех элементов.
  2. Второй цикл: проходит позиции 1, 2, ..., n-1 (это точки раздела). На каждой позиции j:
    • Добавляет элемент a[j-1] к сумме левой части.
    • Вычисляет сумму правой части как total - left.
    • Если они равны, возвращает 1.
  3. Возврат: если ни одна позиция не подошла, функция возвращает 0.
  4. Сложность: O(n), так как два цикла работают за линейное время.
ВНИМАНИЕ НА ГРАНИЦЫРаздел должен дать две непустые части. Поэтому j начинается с 1 (левая часть минимум с одним элементом) и заканчивается перед n (правая часть минимум с одним элементом).

Сложение чисел длины N

Складываем два больших чисела, представленные как массивы десятичных разрядов, справа налево с переносом.

Условие: Два массива целых чисел (элементы от 0 до 9) представляют две последовательности цифр длины N. Реализовать сложение этих чисел. Результат имеет длину N+1 (на случай переноса в старший разряд).

Схема алгоритма

Два числа складываются справа налево (от младшего разряда к старшему) с накоплением переноса.

ПозицияОписание
Справа (индекс n-1)Складываем единицы
Влево (индекс n-2, ...)Складываем десятки, сотни и т.д., плюс перенос
Слева (индекс 0)Записываем последний перенос в дополнительную ячейку

Пример трассировки

Входные данные: 3, затем [1, 2, 3] и [1, 3, 7]. Ожидаемый результат: [0, 2, 6, 0].

Смысл: число 123 + 137 = 260. В массиве результата это [0, 2, 6, 0].

ИндексДействиеa[i]b[i]sumresult[i+1]carry
i=2Единицы373+7+0=1001
i=1Десятки232+3+1=660
i=0Сотни111+1+0=220

Полная программа

#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>

static void add_digits(const int *a, const int *b, int n, int *result)
{
    int carry = 0;
    /* В индексе n-1 находится разряд единиц, складываем справа налево. */
    for (int i = n - 1; i >= 0; --i) {
        int sum = a[i] + b[i] + carry;
        result[i + 1] = sum % 10;
        carry = sum / 10;
    }
    result[0] = carry; /* В старшую ячейку записываем последний перенос. */
}

int main(void)
{
    FILE *fin = fopen("input.txt", "r");
    if (fin == NULL) return -1;
    FILE *fout = fopen("output.txt", "w");
    if (fout == NULL) { fclose(fin); return -1; }

    int status = -1, n = 0;
    int *a = NULL, *b = NULL, *result = NULL;
    if (fscanf(fin, "%d", &n) != 1 || n <= 0) goto done;
    if ((size_t)n > SIZE_MAX / sizeof *a - 1) goto done;
    a = malloc((size_t)n * sizeof *a);
    b = malloc((size_t)n * sizeof *b);
    result = calloc((size_t)n + 1, sizeof *result);
    if (a == NULL || b == NULL || result == NULL) goto done;

    for (int i = 0; i < n; ++i)
        if (fscanf(fin, "%d", &a[i]) != 1 || a[i] < 0 || a[i] > 9) goto done;
    for (int i = 0; i < n; ++i)
        if (fscanf(fin, "%d", &b[i]) != 1 || b[i] < 0 || b[i] > 9) goto done;

    add_digits(a, b, n, result);
    for (int i = 0; i < n + 1; ++i)
        if (fprintf(fout, "%d%c", result[i], i == n ? '\n' : ' ') < 0) goto done;
    status = 0;

done:
    free(a); free(b); free(result);
    if (fclose(fin) != 0) status = -1;
    if (fclose(fout) != 0) status = -1;
    return status;
}

Пошаговое объяснение

  1. Три массива: a и b хранят две последовательности цифр, result (длины n+1) — результат. Индекс 0 — старший разряд.
  2. Инициализация переноса: carry = 0 перед циклом.
  3. Цикл с конца: for (int i = n - 1; i >= 0; --i) проходит от младшего разряда (единиц) к старшему.
  4. Сумма и модуль: sum = a[i] + b[i] + carry может быть до 9 + 9 + 1 = 19. result[i+1] = sum % 10 даёт цифру результата, carry = sum / 10 даёт новый перенос (0 или 1).
  5. Старший разряд: после цикла result[0] = carry. Это может быть 0 (если суммы не произошло переполнения) или 1 (если произошло).
  6. Вывод: все n+1 элементов пишутся в файл, разделённые пробелом (кроме последнего).
ПРАВИЛО ИНДЕКСОВЕсли в исходных массивах a и b индекс 0 — это самый старший разряд, то i идёт от n-1 (младший) к 0 (старший). Результат записывается в result[i+1], поэтому значения сдвигаются на 1 вправо, а result[0] остаётся для финального переноса.
